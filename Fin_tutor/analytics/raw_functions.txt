def last_month():
    # Заработок за последний месяц
    end_date = timezone.now()
    start_date = end_date - timedelta(days=30)
    earning_last_month = Event.objects.filter(date__gte=start_date, date__lte=end_date, is_payed=True).aggregate(

def average_hours_per_week_last_3_months():
    # Среднее количество часов работы в неделю за последние 3 месяца
    end_date = timezone.now()
    start_date = end_date - timedelta(days=90)
    total_hours_worked = Event.objects.filter(date__gte=start_date, date__lte=end_date, is_payed=True).aggregate(
        total_length=Sum('length'))
    weeks = (end_date - start_date).days / 7
    return total_hours_worked['total_length'] / weeks
